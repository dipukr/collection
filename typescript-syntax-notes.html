2. Variables & Basic Types

Annotations go after the name. Most of the time you can leave them off and let inference do the work — annotate function parameters and public API boundaries, not every local variable. 

let a: string = "hi";        // explicit annotation
const b = 42;                // inferred: 42 (literal type)
let c = true;                // inferred: boolean

let s: string   = "text";
let n: number   = 3.14;      // int, float, NaN, Infinity — all `number`
let big: bigint = 100n;
let sym: symbol = Symbol("key");
let u: undefined = undefined;
let nul: null    = null;

any vs unknown

Both accept any value. The difference is what you can do next: any disables checking, unknown forces you to prove the type first. 

let x: any = getData();
x.foo.bar();                  // allowed — may crash at runtime

let y: unknown = getData();
// y.foo;                     // Error: y is of type 'unknown'
if (typeof y === "string") y.toUpperCase();   // OK after narrowing
Reach for unknown at boundaries you do not control — JSON, form input, caught errors. Reach for any almost never.

never

The type with no values. It marks code paths that cannot return.

function fail(msg: string): never {
    throw new Error(msg);
}

function loop(): never {
    while (true) {}
}

type Impossible = string & number;   // never

Strict null checks

With strictNullChecks on, null and undefined are no longer assignable everywhere, so nullable values must say so. 

let name: string = null;        // Error under strictNullChecks
let maybe: string | null = null; // OK
let opt: string | undefined;     // OK

3. Arrays & Tuples

let nums: number[] = [1, 2, 3];
let strs: Array<string> = ["a"];              // generic form
let matrix: number[][] = [[1, 2], [3, 4]];
let mixed: (string | number)[] = [1, "a"];
let frozen: readonly number[] = [1, 2];

readonly arrays

const list: readonly number[] = [1, 2, 3];
// list.push(4);              // Error — no mutating methods
const copy = [...list, 4];    // OK

Tuples — fixed length, meaningful positions

let pair: [string, number] = ["age", 30];
let named: [x: number, y: number] = [1, 2];      // labeled elements
let opt: [string, number?] = ["a"];              // optional element
let rest: [string, ...number[]] = ["a", 1, 2];   // rest element
let ro: readonly [number, number] = [0, 0];

const [key, value] = pair;    // destructuring preserves types

Tuples are what make patterns like useState work: position 0 is the value, position 1 is the setter, and each gets its own type.

4. Object Types

let user: { name: string; age: number } = { name: "Ada", age: 36 };

let config: {
    host: string;
    port?: number;         // optional
    readonly id: string;   // cannot be reassigned
} = { host: "localhost", id: "a1" };

Index signatures

For objects whose keys are not known ahead of time.

type Dict = { [key: string]: number };
type NumKeys = { [key: number]: string };

const scores: Dict = { math: 90, art: 85 };
scores.science = 70;          // OK

// Record is usually cleaner
type Dict2 = Record<string, number>;
type Roles = Record<"admin" | "user", string[]>;

Excess property checks

A fresh object literal may not carry properties the target type does not declare. Assign through a variable and the check no longer applies — this catches typos, not structural mismatches. 

type P = { x: number };

const p: P = { x: 1, y: 2 };  // Error: 'y' does not exist on type 'P'

const raw = { x: 1, y: 2 };
const p2: P = raw;            // OK — not a fresh object literal

5. Interfaces

interface User {
    readonly id: number;
    name: string;
    email?: string;
    greet(): string;                    // method
    update: (name: string) => void;     // function-typed property
}

interface Animal { name: string }
interface Dog extends Animal { breed: string }

Declaration merging

Two interfaces with the same name combine, which is how library types get extended.

interface Window { title: string }
interface Window { version: number }
// Window now has both members

Call and construct signatures

interface Formatter {
    (input: string): string;      // call signature
    locale: string;               // ...plus a property
}

interface DateCtor {
    new (iso: string): Date;      // construct signature
}

6. Type Aliases

A name for any type, including ones interfaces cannot express.

type ID = string | number;
type Point = { x: number; y: number };
type Callback = (err: Error | null, data?: string) => void;
type Pair<T> = [T, T];
type Tree<T> = { value: T; children: Tree<T>[] };   // recursive

type Json =
    | string | number | boolean | null
    | Json[]
    | { [k: string]: Json };

7. Unions & Intersections

// Union — one of
type Status = "idle" | "loading" | "done";

function print(x: string | number) {
    console.log(x.toString());   // only shared members are available
}

// Intersection — all of
type Named = { name: string };
type Aged  = { age: number };
type Person = Named & Aged;

const p: Person = { name: "Ada", age: 36 };

On a union you can only touch members that exist on every branch, which is why narrowing (section 12) matters so much.

Discriminated unions

A shared literal field acts as a tag the compiler can switch on. This is the single most useful modelling pattern in TypeScript — it makes illegal states unrepresentable and gives you exhaustiveness checking for free. 

type Shape =
    | { kind: "circle"; radius: number }
    | { kind: "square"; side: number }
    | { kind: "rect"; w: number; h: number };

function area(s: Shape): number {
    switch (s.kind) {
        case "circle": return Math.PI * s.radius ** 2;
        case "square": return s.side ** 2;
        case "rect":   return s.w * s.h;
        default:
            const _exhaustive: never = s;   // compile error if a case is missed
            return _exhaustive;
    }
}
The never assignment in default turns a forgotten case into a compile error the moment a new shape is added to the union.

8. Literal Types & Enums

let dir: "left" | "right";
let dice: 1 | 2 | 3 | 4 | 5 | 6;

const cfg  = { mode: "dark" };            // { mode: string }
const cfg2 = { mode: "dark" } as const;   // { readonly mode: "dark" }

const ROLES = ["admin", "user", "guest"] as const;
type Role = typeof ROLES[number];         // "admin" | "user" | "guest"

Enums

enum Direction { Up, Down, Left, Right }         // 0, 1, 2, 3
enum Status { Active = "ACTIVE", Banned = "BANNED" }
const enum Fast { A, B }                         // inlined, no runtime object

Direction.Up;     // 0
Direction[0];     // "Up"  (reverse mapping, numeric enums only)

Enums are one of the few TypeScript features that emit runtime code. The const-object pattern below gives you the same ergonomics with plain JavaScript output, and it works under erasableSyntaxOnly and Node's native type stripping. 

const Status = { Active: "ACTIVE", Banned: "BANNED" } as const;
type Status = typeof Status[keyof typeof Status];   // "ACTIVE" | "BANNED"

9. Functions

function add(a: number, b: number): number { return a + b; }

const sub = (a: number, b: number): number => a - b;

type BinOp = (a: number, b: number) => number;
const div: BinOp = (a, b) => a / b;

Optional, default, rest & destructured parameters

function greet(name: string, greeting = "Hello", punct?: string) {
    return `${greeting}, ${name}${punct ?? "!"}`;
}

function sum(...nums: number[]): number {
    return nums.reduce((a, b) => a + b, 0);
}

function draw({ x, y, color = "red" }: { x: number; y: number; color?: string }) {}

Optional parameters must come after required ones, and ? cannot be combined with a default value — a default already makes the argument optional.

Overloads

function parse(x: string): number;
function parse(x: number): string;
function parse(x: string | number): number | string {
    return typeof x === "string" ? Number(x) : String(x);
}
// Only the two overload signatures are callable; the third is the implementation.

Type predicates & assertion signatures

These teach the compiler about checks it cannot see through on its own.

function isString(x: unknown): x is string {
    return typeof x === "string";
}

function assertDefined<T>(x: T): asserts x is NonNullable<T> {
    if (x == null) throw new Error("nullish");
}

function assertOk(cond: unknown): asserts cond {
    if (!cond) throw new Error("assertion failed");
}

Typing this

function handler(this: HTMLElement, ev: Event) {
    this.classList.add("clicked");
}

10. Classes

class Person {
    name: string;
    age = 0;                     // inferred number
    readonly id: string;
    static count = 0;
    #secret = "private";         // native ECMAScript private field

    constructor(name: string, id: string) {
        this.name = name;
        this.id = id;
        Person.count++;
    }

    greet(): string { return `Hi ${this.name}`; }

    get label(): string { return this.name.toUpperCase(); }
    set label(v: string) { this.name = v.toLowerCase(); }

    static create(n: string) { return new Person(n, crypto.randomUUID()); }
}

Access modifiers & parameter properties

class Account {
    public owner!: string;      // default
    private balance = 0;        // compile-time only
    protected bank = "ACME";    // visible to subclasses
    readonly created = new Date();
}

// Parameter properties — declare and assign in one step
class Point {
    constructor(
        public x: number,
        public y: number,
        private readonly label = "pt",
    ) {}
}
ModifierEffect
publicVisible everywhere. The default.
privateSame class only. Compile-time; still visible at runtime.
#fieldSame class only, enforced by the JavaScript runtime.
protectedSame class and subclasses.
readonlyAssignable in the constructor, not afterwards.
staticLives on the class, not on instances.

Inheritance

class Animal {
    constructor(protected name: string) {}
    speak(): string { return `${this.name} makes a noise`; }
}

class Dog extends Animal {
    constructor(name: string, public breed: string) {
        super(name);                  // required before using `this`
    }
    override speak(): string { return `${this.name} barks`; }
}

With noImplicitOverride enabled, the override keyword becomes mandatory — so renaming a base method surfaces every stale subclass immediately.

Abstract classes

abstract class Shape {
    abstract area(): number;
    abstract readonly name: string;
    describe() { return `${this.name}: ${this.area()}`; }
}

class Circle extends Shape {
    readonly name = "circle";
    constructor(private r: number) { super(); }
    area() { return Math.PI * this.r ** 2; }
}

implements only checks conformance. It never inherits members or default implementations, so a class with implements must declare everything itself.

Class types: instance vs constructor

type Instance = Person;                       // instance type
type Ctor     = typeof Person;                // constructor type
type Ctor2    = new (...args: any[]) => Person;

function make<T>(C: new () => T): T { return new C(); }

The this type

class Builder {
    private parts: string[] = [];
    add(p: string): this { this.parts.push(p); return this; }
}
class Sub extends Builder {
    extra(): this { return this; }
}
new Sub().add("a").extra();     // chaining survives subclassing

11. Generics

Type parameters let one definition work across many types without losing information.

function identity<T>(x: T): T { return x; }

identity<string>("a");     // explicit
identity("a");             // inferred

class Box<T> {
    constructor(private value: T) {}
    get(): T { return this.value; }
    map<U>(fn: (v: T) => U): Box<U> { return new Box(fn(this.value)); }
}

Constraints and defaults

function merge<A extends object, B extends object>(a: A, b: B): A & B {
    return { ...a, ...b };
}

function len<T extends { length: number }>(x: T): number { return x.length; }

function pluck<T, K extends keyof T>(obj: T, key: K): T[K] {
    return obj[key];
}

interface Response<T = unknown> { data: T; status: number }   // default param

const type parameters & variance

function names<const T extends readonly string[]>(x: T): T { return x; }
const n = names(["a", "b"]);        // readonly ["a", "b"], no `as const` needed

interface Producer<out T> { get(): T }         // covariant
interface Consumer<in T> { set(v: T): void }   // contravariant
Choosing between the three tools: a union parameter when the body treats every input the same way, a generic when the return type depends on the input type, and overloads when distinct input shapes map to distinct return types.

12. Narrowing & Guards

Every way to convince the compiler a value is more specific than it was declared:

if (typeof x === "string") { }      // typeof
if (x) { }                          // truthiness
if (x != null) { }                  // excludes null and undefined
if (err instanceof TypeError) { }   // instanceof
if ("radius" in shape) { }          // in operator
if (action.type === "add") { }      // discriminant property
if (Array.isArray(x)) { }           // built-in predicate
if (isUser(x)) { }                  // custom predicate
assertDefined(x);                   // assertion signature

Control-flow analysis

Narrowing follows the flow of the code, including early returns.

function f(x: string | number | null) {
    if (x === null) return;                     // x: string | number
    if (typeof x === "number") return x.toFixed(2);
    return x.trim();                            // x: string
}

Caught errors

try {
    risky();
} catch (e: unknown) {          // unknown by default under `strict`
    if (e instanceof Error) console.error(e.message);
    else console.error(String(e));
}

13. Assertions & satisfies

const el  = document.getElementById("x") as HTMLInputElement;
const n   = "42" as unknown as number;    // double assertion
const val = maybeNull!;                   // non-null assertion

class C {
    name!: string;      // definite assignment assertion
}
Assertions only silence the compiler. Nothing is checked at runtime, so every as is a claim you are personally guaranteeing.

satisfies — validate without widening

type Config = Record<string, string | number>;

const cfg = {
    host: "localhost",
    port: 8080,
} satisfies Config;

cfg.port.toFixed();    // port stays `number`
// With `const cfg: Config = ...`, port would widen to `string | number`.

The four ways to type a literal, and what each one keeps:

const a = { x: 1 };                          // { x: number }
const b = { x: 1 } as const;                 // { readonly x: 1 }
const c: { x: number } = { x: 1 };           // checked, widened
const d = { x: 1 } satisfies { x: number };  // checked, narrow type kept

14. keyof, typeof & Indexed Access

The three operators that let types be derived from other types rather than restated.

type User = { id: number; name: string; tags: string[] };

type Keys = keyof User;            // "id" | "name" | "tags"
type Id   = User["id"];            // number
type Vals = User[keyof User];      // number | string | string[]
type Tag  = User["tags"][number];  // string

const user = { id: 1, name: "a" };
type U   = typeof user;                      // { id: number; name: string }
type Ret = ReturnType<typeof someFunction>;

15. Conditional Types & infer

An if statement for types.

type IsString<T> = T extends string ? true : false;

type A = IsString<"x">;      // true
type B = IsString<number>;   // false

infer — capturing a type from a shape

type ElementOf<T> = T extends (infer U)[] ? U : never;
type Unwrap<T>    = T extends Promise<infer U> ? U : T;
type Params<T>    = T extends (...args: infer P) => any ? P : never;
type Ret<T>       = T extends (...args: any) => infer R ? R : never;
type First<T>     = T extends [infer F, ...any[]] ? F : never;
type Last<T>      = T extends [...any[], infer L] ? L : never;

Distribution over unions

A naked type parameter in a conditional type is applied to each union member separately. Wrapping both sides in a tuple switches that off. 

type Dist<T>    = T extends any ? T[] : never;
type NonDist<T> = [T] extends [any] ? T[] : never;   // tuple blocks distribution

type X = Dist<string | number>;      // string[] | number[]
type Y = NonDist<string | number>;   // (string | number)[]

16. Mapped Types

Transform every property of a type at once.

type Optional<T>  = { [K in keyof T]?: T[K] };
type Frozen<T>    = { readonly [K in keyof T]: T[K] };
type Stringify<T> = { [K in keyof T]: string };

// Removing modifiers with `-`
type Mutable<T>  = { -readonly [K in keyof T]: T[K] };
type Required2<T> = { [K in keyof T]-?: T[K] };

Key remapping with as

type Getters<T> = {
    [K in keyof T as `get${Capitalize<string & K>}`]: () => T[K]
};
// { name: string }  →  { getName: () => string }

// Mapping a key to `never` removes it
type OmitByKey<T, K extends keyof T> = {
    [P in keyof T as P extends K ? never : P]: T[P]
};

type PickByType<T, V> = {
    [K in keyof T as T[K] extends V ? K : never]: T[K]
};

Recursive variants

type DeepReadonly<T> = T extends (infer U)[]
    ? readonly DeepReadonly<U>[]
    : T extends object
        ? { readonly [K in keyof T]: DeepReadonly<T[K]> }
        : T;

type DeepPartial<T> = T extends object
    ? { [K in keyof T]?: DeepPartial<T[K]> }
    : T;

17. Template Literal Types

String interpolation, in the type system.

type Greeting = `hello ${string}`;
type Lang   = "en" | "fr";
type Locale = `${Lang}-US`;                 // "en-US" | "fr-US"
type CssVar = `--${string}`;
type Route  = `/users/${number}`;

type EventName<T extends string> = `on${Capitalize<T>}`;
type E = EventName<"click">;                // "onClick"

// Intrinsic string types
type A = Uppercase<"hello">;      // "HELLO"
type B = Lowercase<"HELLO">;      // "hello"
type C = Capitalize<"hello">;     // "Hello"
type D = Uncapitalize<"Hello">;   // "hello"

Parsing strings with infer

type Split<S extends string, D extends string> =
    S extends `${infer Head}${D}${infer Tail}`
        ? [Head, ...Split<Tail, D>]
        : [S];

type Parts = Split<"a.b.c", ".">;    // ["a", "b", "c"]

18. Built-in Utility Types

type User = { id: number; name: string; email?: string };

type Draft  = Partial<User>;              // all optional
type Full   = Required<User>;             // email becomes required
type Frozen = Readonly<User>;
type Public = Pick<User, "id" | "name">;
type NoId   = Omit<User, "id">;
type ById   = Record<string, User>;

type A = Exclude<"a" | "b" | "c", "a">;      // "b" | "c"
type B = Extract<string | number, number>;   // number
type C = NonNullable<string | null>;         // string

function f(a: string, b: number): boolean { return true; }
type P = Parameters<typeof f>;    // [string, number]
type R = ReturnType<typeof f>;    // boolean
type I = InstanceType<typeof Date>;          // Date
type W = Awaited<Promise<Promise<string>>>;  // string
UtilityDoes
Partial / RequiredMake every property optional / required.
ReadonlyMake every property readonly.
RecordBuild an object type from keys and a value type.
Pick / OmitKeep or drop the listed keys.
Exclude / ExtractFilter union members out or in.
NonNullableRemove null and undefined.
Parameters / ReturnTypeRead a function's parameters or result.
ConstructorParameters / InstanceTypeSame, for classes.
AwaitedUnwrap a promise, however deeply nested.
ThisParameterType / OmitThisParameterRead or strip the this parameter.
NoInferStop inference from using this position.
Uppercase / Lowercase / Capitalize / UncapitalizeTransform string literal types.

19. Modules & Imports

// Exporting
export const x = 1;
export function f() {}
export default function main() {}
export { a, b as c };
export * from "./mod";
export * as ns from "./mod";
export type { User };

// Importing
import main from "./mod";              // default
import { a, b } from "./mod";          // named
import { a as alias } from "./mod";    // renamed
import * as ns from "./mod";           // namespace
import "./side-effect";                // side-effect only
const mod = await import("./lazy");    // dynamic

Type-only imports

These vanish entirely from the output. They are required when verbatimModuleSyntax or isolatedModules is on and the import is used only in type positions. 

import type { User } from "./types";            // fully erased at compile time
import { type User, createUser } from "./api";  // inline type modifier
export type { User };

import data from "./data.json" with { type: "json" };   // import attributes

Path aliases

{
  "compilerOptions": {
    "baseUrl": ".",
    "paths": {
      "@/*": ["src/*"],
      "@lib": ["src/lib/index.ts"]
    }
  }
}

20. Declaration Files & Ambient Types

A .d.ts file describes types for code that already exists elsewhere.

// globals.d.ts
declare const VERSION: string;
declare function gtag(...args: any[]): void;

declare global {
    interface Window { myApp: { version: string } }
    namespace NodeJS {
        interface ProcessEnv { API_KEY: string }
    }
}

Untyped packages, asset imports, augmentation

declare module "untyped-lib" {
    export function doThing(x: string): number;
}

declare module "*.svg" {
    const content: string;
    export default content;
}

declare module "*.css";     // shorthand — everything is `any`

// Augmenting an existing module
import "express";
declare module "express" {
    interface Request { user?: { id: string } }
}

21. Namespaces

The pre-modules way to group code. Prefer ES modules in new code, but namespaces still appear throughout declaration files and older libraries. 

namespace Geometry {
    export interface Point { x: number; y: number }
    export function dist(a: Point, b: Point): number {
        return Math.hypot(a.x - b.x, a.y - b.y);
    }
    const internal = 1;      // not exported
}

Geometry.dist({ x: 0, y: 0 }, { x: 1, y: 1 });

namespace A.B.C { export const x = 1; }   // nested shorthand
import G = Geometry;                      // alias

22. Async, Promises & Iterators

async function load(): Promise<string> { return "data"; }

async function main() {
    const data: string = await load();
    const [a, b] = await Promise.all([load(), load()]);
}

const [r] = await Promise.allSettled([load()]);
if (r.status === "fulfilled") r.value;   // string
else r.reason;

Generators and iterables

Generator takes three type parameters: what it yields, what it returns, and what may be passed back in via next(). 

function* counter(): Generator<number, void, unknown> {
    let i = 0;
    while (true) yield i++;
}

async function* stream(): AsyncGenerator<string> {
    yield "chunk";
}

for await (const chunk of stream()) console.log(chunk);

class Range implements Iterable<number> {
    constructor(private from: number, private to: number) {}
    *[Symbol.iterator](): Iterator<number> {
        for (let i = this.from; i < this.to; i++) yield i;
    }
}

23. Decorators

Standard decorators need no compiler flag from TypeScript 5.0 onward.

function logged<T extends (...args: any[]) => any>(
    target: T,
    ctx: ClassMethodDecoratorContext,
) {
    return function (this: unknown, ...args: any[]) {
        console.log(`calling ${String(ctx.name)}`);
        return target.call(this, ...args);
    } as T;
}

class Service {
    @logged
    fetch(id: string) { return id; }

    accessor value = 0;      // auto-accessor, also decoratable
}

Context kinds: class, method, getter, setter, field, accessor.

Legacy experimental decorators

Angular, NestJS and TypeORM still use this older form, which is flag-gated.

// tsconfig: { "experimentalDecorators": true, "emitDecoratorMetadata": true }

@Component({ selector: "app" })
class AppComponent {
    @Input() name!: string;

    constructor(@Inject(TOKEN) private svc: Service) {}

    @HostListener("click", ["$event"])
    onClick(e: Event) {}
}

Two options worth understanding

// noUncheckedIndexedAccess

27. Patterns & Idioms

Branded types — nominal typing on demand

TypeScript compares types structurally, so every ID is just a string. A brand makes them distinguishable. 

type Brand<T, B> = T & { readonly __brand: B };
type UserId = Brand<string, "UserId">;
type PostId = Brand<string, "PostId">;

const toUserId = (s: string) => s as UserId;

function getUser(id: UserId) {}
// getUser("raw");        // Error — plain strings and PostIds are rejected

Result types instead of throwing

Errors become part of the signature, and the compiler makes you handle them.

type Result<T, E = Error> =
    | { ok: true;  value: T }
    | { ok: false; error: E };

function parse(s: string): Result<number> {
    const n = Number(s);
    return Number.isNaN(n)
        ? { ok: false, error: new Error("bad number") }
        : { ok: true, value: n };
}

const r = parse("1");
if (r.ok) r.value.toFixed();    // narrowed to number

Small helpers worth keeping around

// Flatten a type so editors show a readable object on hover
type Prettify<T> = { [K in keyof T]: T[K] } & {};

// Exactly one of two prop shapes
type XOR<A, B> =
    | (A & { [K in Exclude<keyof B, keyof A>]?: never })
    | (B & { [K in Exclude<keyof A, keyof B>]?: never });

// Type-safe Object.keys
function keys<T extends object>(o: T): (keyof T)[] {
    return Object.keys(o) as (keyof T)[];
}

// Suggest known values, still accept any string
type LooseAutocomplete<T extends string> = T | (string & {});

type UnionToIntersection<U> =
    (U extends any ? (k: U) => void : never) extends (k: infer I) => void
        ? I : never;

Key Takeaway

TypeScript adds no runtime behaviour.

Every construct here either describes JavaScript that already exists, or disappears at compile time.

The exceptions are worth memorising, because they are the only features that emit code: enum, namespace, parameter properties, and legacy decorators. Everything else — types, interfaces, generics, guards, assertions — is erased.
