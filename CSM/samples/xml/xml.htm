  8|   return t.replace('&',"&amp;",'<',"&lt;",'>',"&gt;",'"',"&quot;",'\'',"&apos;");

 41| //| simple element has only text value, no children

 43| class simple_element: node {
 44|     var     _name;
 45|     var     _attributes;
 46|    
 47|     function simple_element(name,atts,content) {
 48|       node(content);
 49|       _name = name;
 50|       _attributes = atts;
 51|     }
 52| //|    
 53| //| access to children in form of property
 54| //|
 55|     property children() {  /* nothing here */ return null; }
 56| //| 
 57| //| access to attributes 
 58| //| e.g. :  a = xnode["name"];  
 59| //|
 60|     function [](n,v) 
 61|     {
 62|       if(!is_undefined(v)) // set 
 63|       {
 64|         if(!_attributes) _attributes = new map();
 65|         _attributes[n] = v;
 66|         return v;
 67|       } 
 68|       if(_attributes) return _attributes[n];
 69|       return null;
 70|     }
 71| //|    
 72| //| value here is just all text children
 73| //| always returns string
 74| //|
 75|     property value(v) {  
 76|     // read/write  
 77|       if(!is_undefined(v)) _value = v;
 78|       return _value;
 79|     }
 80| //|    
 81| //| creates string representation of content
 82| //| always returns string
 83| //|
 84|     function valueToString() {  
 85|       //out.printf("DEBUG: (%s) valueToString\n",_name);
 86|       if(_value) return unescape(_value);
 87|       return "";
 88|     }
 89|   
 90|     function toString() {
 91|       var i;
 92|       var s = "<" + _name;
 93|       if(_attributes) 
 94|           for (i = 0; i < _attributes.length; i++) 
 95|             s += string::printf(" %s=\"%s\"",_attributes.key(i),unescape(_attributes.value(i)));
 96|       if(_value)
 97|         s += (" >" + valueToString() + string::printf("</%s>", _name));
 98|       else 
 99|         s += " />";  
100|       return s;
101|     }
102| }

105| //| element has text value(s) and/or children 
106| //|
107| class element: simple_element 
108| {
109|   
110|     function element(name,atts,content) { simple_element(name,atts,content); }
111| //|    
112| //| value here is just all text children
113| //| always returns string
114| //|
115|     property value() {  
116|       // read only 
117|       if(!_value) return "";
118|       var i,v = ""; 
119|       for(i = 0; i < _value.length; i++) if(is_string(_value[i])) v += _value[i];
120|       return v;
121|     }
122| //|    
123| //| returns first element with the given tag_name
124| //|
125|     function child(tag_name)
126|     {
127|       if(_value) {
128|         var i; 
129|         for(i = 0; i < _value.length; i++) 
130|           if( (_value[i] instanceof xml::simple_element) && 
131|               (_value[i]._name == tag_name) ) return _value[i];
132|       }
133|       return null;
134|     }
135| //|    
136| //| tag_name here is template of tag name.
137| //| returns 1) array of subelements with the given tag_name
138| //|      or 2) map of subelements with the given tag_name and having attribute att_name.
139| //|            key of the map is value of att_name (these values supposed to be unique
140| //|            among all children of given element)
141| //|
142|     function elements(tag_name,att_name)
143|     {
144|       var r;
145|       if(att_name) // map
146|       { 
147|         r = new map();
148|         if(!_value) return r; 
149|         var i; 
150|         for(i = 0; i < _value.length; i++) 
151|         {
152|            var el = _value[i];
153|            if( el instanceof xml::simple_element && el._name.like(tag_name)) 
154|            {
155|               var a = el[att_name];
156|               if( a ) r[a] = el;
157|            }    
158|         }   
159|       } else {  // array
160|         r = new array();
161|         if(!_value) return r; 
162|         var i; 
163|         for(i = 0; i < _value.length; i++) 
164|         {
165|            var el = _value[i];
166|            if( el instanceof simple_element && el._name.like(tag_name)) 
167|               r.push(el);
168|         }   
169|       }
170|       return r;
171|     }
172|     function valueToString() {  
173|       if(_value) 
174|       {
175|         var i,v = ""; 
176|         for(i = 0; i < _value.length; i++)
177|           v += is_string(_value[i])? unescape(_value[i]): string(_value[i]);
178|         return v;
179|       }
180|       return "";
181|     }
182| }

188| class parser {
189|   var _in;
190|   var _error_msg;
191|   var _error_ln;
192|   var _got_eot;
193| // vocabulary for interning nmtokens
194|   static var voc = new map();
195|   
196|   function parser() { _in = null; }
197|   
198|   function intern(nmtoken) {
199|       if( !voc.exist(nmtoken) ) voc[nmtoken] = nmtoken;
200|       return voc[nmtoken];
201|   }
202|   
203|   function compile_atts() {
204|     var atts;
205|     while(true) {
206|       var n = _in.get('=',"/>",">").trim(); 
207|       if(_in.get_match == 2) _got_eot = true;
208|       if(_in.get_match != 1) break;
209|       
210|       if(!n) throw "empty name encountered";
211|       
212|       var z = _in.get('\"','\'',"/>",">").trim(); 
213|       if( z ) throw string::printf("bad attribute definition (%s)", n ); 
214|       if(_in.get_match > 2) throw "empty value encountered";
215|       var v = _in.get( _in.get_match == 1 ? '\"' : '\'' ); 
216|       if(!atts) atts = new map();
217|       atts[ intern(n) ] = unescape(v);
218|     }
219|     return atts;
220|   }
221|   
222|   function compile_tag(tag_name)
223|   {
224|     //NB already got '<'
225|     var c = _in.get(); // get one char
226|     switch(c) 
227|     {
228|       case '/': 
229|         { 
230|           var n = _in.get('>').trim(); // get until (exclusive) 
231|           if(tag_name != n) throw string::printf("tag '%s' not closed (%s)",tag_name,n);
232|           return null;
233|         } 
234|       case '?':
235|         { 
236|           var v = _in.get("?>"); 
237|           return new pi(v); 
238|         } 
239|       case '!': 
240|         {
241|           c = _in.get();
242|           if ( c == '-' )
243|           {
244|             if(_in.get() != '-') throw string::printf("bad comment: %s",_in.get("\n"));
245|             var v = _in.get("-->");
246|             return new comment(v);
247|           }
248|           else if ( c == '[' ) {
249|             var v = _in.get("CDATA[","\n");
250|             if(v) throw string::printf("expecting:'CDATA[', got:%s",v);
251|             v = _in.get("]]>");
252|             return new cdata(v);
253|           }
254|           else if( c == 'D' )
255|           {
256|             //skip doctype definition
257|             _in.get("OCTYPE");
258|             _in.get("]");_in.get(">");
259|             return true;
260|           }
261|           else
262|             throw string::printf("expecting comment or cdata, got '%c'",c);
263|         } break;
264|       default:    
265|         {
266|           var atts_map;
267|           _got_eot = false;
268|           var name = intern( new string(c) + _in.get(' ',"/>",'>').trim() );
269|           if(_in.get_match == 1) atts_map = compile_atts(_in);
270|           else if(_in.get_match == 2) _got_eot = true;
271|           var content;
272|           if(!_got_eot) 
273|             while(!_in.eof) 
274|             {
275|               var v = _in.get('<');
276|               //out.printf("DEBUG: (%s) content (%s)\n",name,content);                        
277|               if(v) 
278|               {
279|                 if(!content) // make it string
280|                   content = unescape(v);  
281|                 else // array
282|                   content.push(unescape(v));
283|               }
284|               v = compile_tag(name);
285|               if(v) {
286|                 if( !is_array(content) ) content = content? new array(content) : new array(); 
287|                 content.push(v);
288|               }
289|               else // this was tail of this tag
290|                 break; 
291|             }
292|           return is_array(content)? 
293|                  new element(name,atts_map,content):
294|                  new simple_element(name,atts_map,content);   
295|         }
296|     }
297|   }
298|   function parse(in_stream)
299|   {
300|       _in = in_stream;
301|       var r = null;      
302|       try {
303|         do {
304|           if( typeof(_in.get('<')) == type::NULL) break;
305|           r = compile_tag(".");
306|           //skiping all pi's and doctypes
307|         } while( !(r instanceof simple_element) ); // or element
308|       } catch (e) { std::out.printf("ERROR:%s",e); }
309|       _in.close();
310|       return r;  
311|   }
312| }
