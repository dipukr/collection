  9| const NUMBER_COLOR = "red";
 10| const KEYWORD_COLOR = "darkblue";
 11| const FUNCTION_COLOR = "purple";
 12| const COMMENT_COLOR = "green";
 13| const STRING_LIT_COLOR = "maroon";
 14| const OPERATOR_COLOR = "dimgray";

 21| var _in = null;
 22| var _out = null;

 24| var _line_no = 0;

 26| var _r_words = {
 27|   "class"   : true ,  "static"  : true ,  "if"      : true ,
 28|   "else"    : true ,  "while"   : true ,  "return"  : true ,
 29|   "for"     : true ,  "break"   : true ,  "continue": true ,
 30|   "do"      : true ,  "new"     : true ,  "null"    : true ,
 31|   "var"     : true ,  "function": true ,  "package" : true ,
 32|   "try"     : true ,  "catch"   : true ,  "throw"   : true ,
 33|   "property": true ,  "import"  : true ,  "is"      : true ,
 34|   "undefined":true ,  "$"       : true ,  "$$"      : true ,
 35|   "switch"  : true ,  "case"    : true ,  "default" : true ,
 36|   "@"       : true ,  "synchronized": true , "const"   : true,
 37|   "true"    : true,   "false" : true
 38| };

 64| function escape(s) {
 65|   return s.replace('&',"&amp;",'<',"&lt;",'>',"&gt;",'"',"&quot;"); 
 66|   // '\'',"&apos;"

 67| }
 68| 
 69| function emit_line_no() {
 70|   _out.printf("\n<FONT color='%s'>%3d| </FONT>",LINE_NO_COLOR,++_line_no);
 71| }
 72| function emit_number(tok) {
 73|   _out.printf("<FONT color='%s'>%s</FONT>",NUMBER_COLOR,tok);
 74| }
 75| function emit_keyword(tok) {
 76|   /*
 77|    out.printf("<FONT color='%s'>%s</FONT>",KEYWORD_COLOR,tok);
 78|    */
 79|   _out.printf("<FONT color='%s'><B>%s</B></FONT>",KEYWORD_COLOR,tok);
 80| }
 81| function emit_call(tok) {
 82|   _out.printf("<FONT color='%s'>%s</FONT>",FUNCTION_COLOR,tok);
 83| }
 84| function emit_plain_id(tok) {
 85|   _out.put(tok);
 86| }
 87| function emit_string(tok) {
 88|   _out.printf("<FONT color='%s'>&quot;%s&quot;</FONT>",STRING_LIT_COLOR,tok);
 89| }
 90| function emit_char(tok) {
 91|   _out.printf("<FONT color='%s'>\'%s\'</FONT>",STRING_LIT_COLOR,tok);
 92| }
 93| 
 94| function emit_whitespace(idx) {
 95|   switch(idx)
 96|   {
 97|     case 1: _out.put(' ');  break;   
 98|     case 2: emit_line_no(); break;
 99|     case 3: break; // '\r'    

100|     case 4: _out.put(TAB);  break;
101|   }
102| }
103| 
104| function emit_operator(idx) {
105|   _out.printf("<FONT color='%s'>%s</FONT>",OPERATOR_COLOR, escape(_r_delimeters[ idx - 1 ]) );
106| }
107| 
108| // string or char literal

109| function parse_sc_literal(dlms) {
110|   var r = ""; 
111|   while(true) {
112|     var tok = _in.get(dlms); 
113|     if( tok == null ) break;
114|     r += tok; 
115|     if( _in.get_match == 2 ) break;
116|     if( _in.get_match == 1 ) r += dlms[0]; 
117|   }  
118|   return escape(r);
119| }
120| 
121| function emit_c_comment() {
122|   _out.printf("<FONT color='%s'>/*",COMMENT_COLOR);  
123|   while( true ) 
124|   {
125|     var tok = _in.get("\n", "\r", "*/");
126|     if( _in.get_match == 0 ) return;
127|     if( _in.get_match == 1 ) 
128|     {
129|       _out.printf("</FONT>"); 
130|       emit_line_no(); 
131|       _out.printf("<FONT color='%s' >%s ",COMMENT_COLOR,escape(tok));  
132|     }
133|     else if( _in.get_match == 2 ) { _out.put(escape(tok)); }  
134|     else if( _in.get_match == 3 ) 
135|     {
136|       _out.put(escape(tok));
137|       break;
138|     }
139|   }
140|   _out.printf("*/</FONT>");  
141| }
142| function emit_cpp_comment() {
143|   var tok = _in.get('\n');
144|   _out.printf("<FONT color='%s'>//%s</FONT>",COMMENT_COLOR,escape(tok));  
145|   emit_line_no();
146| }
147| 
148| function parse() {
149|   var tok;
150|   while(true) {
151|     tok = _in.get(_r_delimeters); 
152|     if( is_null(tok) ) break; //eof

153|         
154|     if( tok.like("[0-9]*") ) emit_number(tok);
155|     else if( _r_words.exist(tok) ) emit_keyword(tok);
156|     else if( _in.get_match == 7 /* '(' */ ) emit_call(tok);
157|     else if( tok ) emit_plain_id(tok);
158|         
159|     if( _in.get_match == 0) break; //this was last tok before eof

160|     
161|     if( _in.get_match <= 4 ) emit_whitespace(_in.get_match);
162|     else if( _in.get_match == 5 /*string lit*/ ) emit_string( parse_sc_literal(["\\\"","\""]) );
163|     else if( _in.get_match == 6 /*char lit*/ ) emit_char( parse_sc_literal(["\\\'","\'"]) );
164|     else if( _in.get_match == 9 /* C comment */ ) emit_c_comment(); 
165|     else if( _in.get_match == 10/* C++ comment */ ) emit_cpp_comment();
166|     else emit_operator(_in.get_match);
167|   }
168| }
169| 
170| function do_parse(nin)
171| {
172|   _line_no = 0;
173|   out.printf("parsing '%s' ...",nin.name_extension);
174|   _in = nin.stream("r");
175|   if(!_in) out.printf("ERROR: cannot open '%s'\n",nin.path);
176|   
177|   var nout = new node(nin.path);
178|   nout.extension = "htm";
179|   _out = nout.stream("w");
180|   if(!_out) out.printf("ERROR: cannot create '%s'\n",nout.path);
181|   
182|   _out.put("<HTML>\n");
183|   _out.put("<HEAD>\n");
184|   _out.put("<META NAME='GENERATOR' Content='TIL:til2html'>\n");  
185|   _out.put("<LINK rel='stylesheet' type='text/css' href='codestyle.css'>\n");
186|   _out.put("</HEAD>\n");
187|   _out.put("<BODY>\n");
188|   _out.printf("<P>plain text version:<A href=\'%s\'>%s</A></P>\n",nin.name_extension,nin.name_extension);
189|   _out.put("<HR>\n<PRE>\n");
190|   emit_line_no();
191|   parse();
192|   _out.put("</PRE></BODY></HTML>\n");
193|   out.put("done\n");
194| }

196| function main(file_name_in) {
197|   var nin = new node(file_name_in);
198|   if( nin.is_folder ) 
199|   {
200|     var i, narray = nin.nodes("*.csp");
201|     for(i = 0; i < narray.length; i++)
202|       if( narray[i].is_file ) do_parse(narray[i]);
203|   } 
204|   else if(nin.is_file) do_parse(nin);
205|   else out.printf("ERROR: cannot find '%s'\n",file_name_in);
206|   
207| }
