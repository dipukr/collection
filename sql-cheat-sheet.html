Creating & Altering Tables

CREATE TABLE employees (
    id          INT PRIMARY KEY,
    name        VARCHAR(100) NOT NULL,
    department  VARCHAR(50),
    salary      DECIMAL(10,2),
    hired_on    DATE DEFAULT (CURRENT_DATE)
);

-- Add / modify / drop a column
ALTER TABLE employees ADD email VARCHAR(120);
ALTER TABLE employees MODIFY salary DECIMAL(12,2);
ALTER TABLE employees DROP COLUMN email;

-- Remove all rows (fast, no WHERE) vs. delete the table entirely
TRUNCATE TABLE employees;
DROP TABLE employees;

Constraints

ConstraintMeaning
NOT NULLColumn cannot store NULL
UNIQUEAll values must differ
PRIMARY KEYUnique + NOT NULL; row identifier
FOREIGN KEYReferences a key in another table
CHECKValues must satisfy a condition
DEFAULTValue used when none is supplied
CREATE TABLE orders (
    order_id    INT PRIMARY KEY,
    emp_id      INT,
    amount      DECIMAL(10,2) CHECK (amount > 0),
    status      VARCHAR(20) DEFAULT 'PENDING',
    FOREIGN KEY (emp_id) REFERENCES employees(id)
);

CRUD Operations

-- CREATE
INSERT INTO employees (id, name, department, salary)
VALUES (1, 'Asha', 'Engineering', 90000);

-- READ
SELECT name, salary FROM employees WHERE department = 'Engineering';
SELECT * FROM employees;

-- UPDATE
UPDATE employees SET salary = salary * 1.10 WHERE department = 'Engineering';

-- DELETE
DELETE FROM employees WHERE id = 1;

Always include a WHERE clause with UPDATE/DELETE unless you intend to affect every row.

Clauses & Keywords

ClauseUse
WHEREFilter rows

SELECT department, COUNT(*) AS headcount, AVG(salary) AS avg_pay
FROM employees
WHERE name LIKE 'A%'
GROUP BY department
HAVING COUNT(*) > 2
ORDER BY avg_pay DESC
LIMIT 5;

Joins

FROM employees e

Views & Indexes

-- A view is a stored, reusable query
CREATE VIEW high_earners AS
SELECT name, salary FROM employees WHERE salary > 100000;

SELECT * FROM high_earners;
DROP VIEW high_earners;

-- Indexes speed up lookups on frequently-searched columns
CREATE INDEX idx_dept ON employees(department);
DROP INDEX idx_dept;

Triggers & Stored Procedures

-- Trigger: runs automatically on an event
CREATE TRIGGER before_insert_emp
BEFORE INSERT ON employees
FOR EACH ROW
SET NEW.name = UPPER(NEW.name);

-- Stored procedure: reusable named block
CREATE PROCEDURE raise_salary(IN dept VARCHAR(50), IN pct DECIMAL(4,2))
BEGIN
    UPDATE employees SET salary = salary * (1 + pct/100)
    WHERE department = dept;
END;

CALL raise_salary('Engineering', 10);

Syntax for triggers/procedures differs notably between MySQL, PostgreSQL (PL/pgSQL), and SQL Server (T-SQL).

SQL Injection (Security)

SQL injection happens when untrusted input is concatenated directly into a query, letting an attacker alter its logic.

-- VULNERABLE: input concatenated into the query string
query = "SELECT * FROM users WHERE name = '" + userInput + "';"
-- Input  '  OR '1'='1  turns the WHERE clause always true.

-- SAFE: use parameterized queries / prepared statements
PREPARE stmt FROM 'SELECT * FROM users WHERE name = ?';
-- the driver binds userInput as data, never as SQL

Prevention

  • Use parameterized queries / prepared statements (never string concatenation).
  • Validate and sanitize all user input.
  • Apply least-privilege database accounts.
  • Use stored procedures and ORMs that parameterize by default.
